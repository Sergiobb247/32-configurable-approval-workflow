# PostgreSQL driver.
# SQLAlchemy 2.1 (released 2026-09-24) changed the default DBAPI for a bare
# `postgresql://` URL from psycopg2 to psycopg (v3) — a DIFFERENT package from
# psycopg2. With only psycopg2-binary installed, a fresh `pip install` resolves
# SQLAlchemy 2.1 and every connection dies with
#     ModuleNotFoundError: No module named 'psycopg'
# Shipping both drivers means the URL works whichever SQLAlchemy version pip
# resolves: 2.x default -> psycopg3, and `postgresql+psycopg2://` -> psycopg2.
psycopg[binary]>=3.1
psycopg2-binary>=2.9
